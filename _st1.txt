544:   /// 把提取到的门店列表合并到门店配置：只按门店ID去重合并（不因账号/后台写法差异产生重复），
545:   /// 保留用户改过的门店名称与勾选状态
546:   Future<void> _applyStores(List<PospalSubStore> stores) async {
547:     if (stores.isEmpty) return;
548:     final baseUrl = _normMasterUrl(_masterBaseUrlCtrl.text);
549:     final account = _masterAccountCtrl.text.trim();
550:     // 1) 按门店ID去重
551:     final seenIds = <String>{};
552:     final uniqueStores = <PospalSubStore>[];
553:     for (final s in stores) {
554:       if (seenIds.contains(s.id)) continue;
555:       seenIds.add(s.id);
556:       uniqueStores.add(s);
557:     }
558:     // 2) 保留旧配置中用户改过的名称与勾选状态（只按 storeId 匹配）
559:     final oldByStoreId = <String, StoreConfig>{};
560:     for (final c in _configs) {
561:       if (c.storeId.isNotEmpty) {
562:         oldByStoreId[c.storeId] = c;
563:       }
564:     }
565:     // 3) 完全按本次同步的门店顺序重建列表（不沿用旧顺序），
566:     //    避免重新同步后门店排序错乱；本次未同步到的旧门店保留在末尾，防止丢失
567:     final keepIds = uniqueStores.map((s) => s.id).toSet();
568:     final newConfigs = <StoreConfig>[];
569:     var added = 0;
570:     var updated = 0;
571:     for (final s in uniqueStores) {
572:       final old = oldByStoreId[s.id];
573:       if (old != null) {
574:         newConfigs
575:             .add(old.copyWith(name: old.name.isNotEmpty ? old.name : s.name));
576:         updated++;
577:       } else {
578:         newConfigs.add(StoreConfig(
579:           name: s.name,
580:           account: account,
581:           baseUrl: baseUrl,
582:           storeId: s.id,
583:           enabled: true,
584:         ));
585:         added++;
586:       }
587:       await widget.sessionManager.saveUserId(
588:         '$baseUrl|$account|${s.id}',
589:         s.id,
590:       );
591:     }
592:     // 3.5) 保留本次未同步到的旧门店（提取不全时不丢失）
593:     for (final c in _configs) {
594:       if (c.storeId.isNotEmpty && !keepIds.contains(c.storeId)) {
595:         newConfigs.add(c);
596:       }
597:     }
598:     // 4) 双保险：再次按门店ID去重
599:     _configs = _dedupeConfigs(newConfigs);
600:     _storeNameCtrls.clear();
601:     await widget.configService.saveConfigs(_configs);
602:     widget.onConfigChanged?.call();
603:     if (mounted) {
604:       setState(() {});
605:       ScaffoldMessenger.of(context).showSnackBar(
606:         SnackBar(
607:           content: Text(
608:               '门店同步完成：共 ${uniqueStores.length} 家门店（新增 $added，更新 $updated），可在下方勾选搜索门店'),
609:           backgroundColor: AppConstants.successColor,
610:         ),
611:       );
612:     }
613:   }
