604:       if (!mounted || _lastResult?.barcode != barcode) return;
605:       setState(() {
606:         _uncheckedStoreProducts

608:           ..addAll(found);

616:   /// 自动照片同步：勾选/未勾选门店中存在“同款商品部分有图、部分缺图”时，
617:   /// 直接加入照片后台队列把图补到缺图门店（无需手动点按钮）。
618:   Future<void> _maybeAutoSyncPhoto(String barcode) async {
619:     if (!mounted || _imageSyncedToAll) return;
620:     final r = _lastResult;
621:     if (r == null || r.barcode != barcode) return;
622:     // 源商品：勾选结果里第一家“有图”的门店；勾选门店都没有图则看未勾选门店
623:     ProductData? src;
624:     String? srcStoreName;
625:     for (final s in r.stores.values) {
626:       if (s.ok && s.data != null && _hasStorePhoto(s.data!)) {
627:         src = s.data;
628:         srcStoreName = s.storeName;
629:         break;
630:       }
631:     }
632:     if (src == null) {
633:       for (final c in widget.configs) {
634:         final p = _uncheckedStoreProducts[c.name];
635:         if (p != null && _hasStorePhoto(p)) {
636:           src = p;
637:           srcStoreName = c.name;
638:           break;
639:         }
640:       }
641:     }
642:     if (src == null) return;
643:     final rawUrl = src.imageUrl ?? '';
644:     if (rawUrl.isEmpty) return;
645:     final full =
646:         rawUrl.startsWith('http') ? rawUrl : 'https://img.pospal.cn$rawUrl';
647:     final original = full.replaceAll('_200x200', '');
648: 
649:     // 目标门店：与源商品同名（同一商品）、但该店缺图的门店（勾选+未勾选）
650:     final missing = <StoreConfig>[];
651:     final enabledByName = <String, StoreConfig>{};
652:     for (final c in widget.configs) {
653:       if (c.enabled) enabledByName[c.name] = c;
654:     }
655:     for (final s in r.stores.values) {
656:       final cfg = enabledByName[s.storeName];
657:       if (cfg == null || cfg.name == srcStoreName) continue;
658:       final d = s.data;
659:       if (d == null || d.name != src.name) continue;
660:       if (!_hasStorePhoto(d)) missing.add(cfg);
661:     }
662:     for (final c in widget.configs) {
663:       if (c.enabled || c.name == srcStoreName) continue;
664:       final p = _uncheckedStoreProducts[c.name];
665:       if (p == null || p.name != src.name) continue;
666:       if (!_hasStorePhoto(p)) missing.add(c);
667:     }
668:     if (missing.isEmpty) return;
669: 
670:     final uid = src.uid?.toString();
671:     if (await PhotoQueueService.instance.hasActiveJob(barcode, uid)) return;
672:     try {
673:       final job = await PhotoQueueService.instance.enqueue(
674:         type: PhotoJobType.sync,
675:         barcode: barcode,
676:         productName: src.name,
677:         productUid: uid,
678:         sourceUrl: original,
679:         writeDesc: false,
680:         stores: missing,
681:       );
682:       _queueWatchBarcode = barcode;
683:       _queueWatchUid = uid;
684:       if (mounted) setState(() => _photoQueuedMark = true);
685:       if (!mounted) return;
686:       _showBanner(
687:           '部分门店缺图，已自动加入后台队列同步照片（${job.stores.length} 个门店，队列剩余 ${PhotoQueueService.instance.pendingCount} 条）');
688:     } catch (_) {
689:       // 入队失败静默处理，不打断查询
690:     }
691:   }
