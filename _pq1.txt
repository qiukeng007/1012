228:   /// 照片提交的目标门店：总账号模式(存在门店ID)取全部有效门店；

231:     final hasMaster = configs.any((c) => c.storeId.isNotEmpty);

238:   // ==================== 入队 ====================
239: 
240:   /// 入队一张照片任务。[imageBytes] 与 [sourceUrl] 二选一：
241:   /// 新照片传字节（本地压缩后落盘）；同步已有照片传 CDN 原图地址（Worker 下载）。
242:   Future<PhotoJob> enqueue({
243:     required PhotoJobType type,
244:     required String barcode,
245:     required String productName,
246:     String? productUid,
247:     List<int>? imageBytes,
248:     String? sourceUrl,
249:     String opName = '',
250:     bool writeDesc = false,
251:     required List<StoreConfig> stores,
252:   }) async {
253:     if (stores.isEmpty) {
254:       throw StateError('没有可同步的有效门店');
255:     }
256:     final code = barcode.trim();
257:     if (code.isEmpty) throw StateError('条码为空');
258:     if ((imageBytes == null) == (sourceUrl == null)) {
259:       throw StateError('必须且只能提供 imageBytes 或 sourceUrl 之一');
260:     }
261: 
262:     // 同商品(条码+uid)合并：丢弃旧任务只跑最新
263:     final oldJobs = await _loadQueue();
264:     for (final dup in oldJobs) {
265:       if (dup.barcode != code ||
266:           (dup.productUid ?? '') != (productUid ?? '') ||
267:           dup.status.isFinished) {
268:         continue;
269:       }
270:       if (dup.status == PhotoJobStatus.processing) {
271:         dup.superseded = true;
272:         _cancelledIds.add(dup.id);
273:         await _saveQueueJob(dup);
274:       } else {
275:         await _deleteImage(dup);
276:         await _deleteQueueFile(dup.id);
277:       }
278:     }
279: 
280:     final now = DateTime.now();
281:     final id =
282:         '${now.millisecondsSinceEpoch}_${(_seq++).toString().padLeft(3, '0')}';
283:     String? fileName;
284:     if (imageBytes != null) {
285:       final compressed = compressForUpload(Uint8List.fromList(imageBytes));
286:       fileName = '$id.jpg';
287:       final file =
288:           File('${(await _imgDir()).path}${Platform.pathSeparator}$fileName');
289:       await file.writeAsBytes(compressed, flush: true);
290:     }
291: 
292:     final job = PhotoJob(
293:       id: id,
294:       type: type,
295:       barcode: code,
296:       productName: productName,
297:       productUid: productUid,
298:       opName: opName,
299:       opDesc: _actionDesc(type),
300:       writeDesc: writeDesc,
301:       createdAt: now,
302:       status: PhotoJobStatus.pending,
303:       imageFile: fileName,
304:       sourceUrl: sourceUrl,
305:       stores: List.of(stores),
306:     );
307:     await _saveQueueJob(job);
308:     await refreshCount();
309:     _notify();
310:     start();
311:     return job;
312:   }

356:   /// 重试一条已放弃/部分成功的任务（需本地图片仍在或有源URL）
357:   Future<String?> retryJob(String id) async {
358:     final jobs = await _loadHistory();
359:     PhotoJob? job;
360:     for (final j in jobs) {
361:       if (j.id == id) {
362:         job = j;
363:         break;
364:       }
365:     }
366:     if (job == null) return '未找到该记录';
367:     _cancelledIds.remove(id);
368:     if (!job.canRetryNow) {
369:       return '该记录的图片已清理，无法自动重试，请重新拍照提交';
370:     }
371:     job.status = PhotoJobStatus.pending;
372:     job.attempts = 0;
373:     job.nextAttemptAt = 0;
374:     job.finishedAt = null;
375:     job.totalMs = null;
376:     job.results = [];
377:     await _deleteHistoryFile(job.id);
378:     await _saveQueueJob(job);
379:     await refreshCount();
380:     _notify();

409:   Future<void> _pumpLoop() async {
410:     while (_pumping) {
411:       final qs = _queryService;
412:       if (qs == null) break;
413:       final jobs = await _loadQueue();
414:       if (jobs.isEmpty) break;
415: 
416:       // 崩溃/杀进程恢复：处理中被中断的任务回到排队重跑；已被取代的收尾归档
417:       for (final j in jobs) {
418:         if (j.superseded) {
419:           await _finishSuperseded(j);
420:         } else if (j.status == PhotoJobStatus.processing) {
421:           j.status = PhotoJobStatus.pending;
422:           j.nextAttemptAt = 0;
423:           await _saveQueueJob(j);
424:         }
425:       }
426: 
427:       final nowMs = DateTime.now().millisecondsSinceEpoch;
428:       PhotoJob? todo;
429:       int? soonest;
430:       for (final j in jobs) {
431:         if (j.status == PhotoJobStatus.pending && j.nextAttemptAt <= nowMs) {
432:           todo = j;
433:           break;
434:         }
435:         if (j.status == PhotoJobStatus.pending && j.nextAttemptAt > nowMs) {
436:           if (soonest == null || j.nextAttemptAt < soonest) {
437:             soonest = j.nextAttemptAt;
438:           }
439:         }
440:       }
441: 
442:       if (todo == null) {
443:         // 无待办：可能在退避，也可能在等待登录（Cookie 恢复后自动转回排队）
444:         var anyWaiting = false;
445:         for (final j in jobs) {
446:           if (j.status != PhotoJobStatus.waitingLogin) continue;
447:           anyWaiting = true;
448:           for (final s in j.stores) {
449:             if (await qs.hasSessionCookie(s)) {
450:               j.status = PhotoJobStatus.pending;
451:               j.nextAttemptAt = 0;
452:               await _saveQueueJob(j);
453:               todo = j;
454:               break;
455:             }
456:           }
457:           if (todo != null) break;
458:         }
459:         if (todo == null) {
460:           final waitMs = soonest != null
